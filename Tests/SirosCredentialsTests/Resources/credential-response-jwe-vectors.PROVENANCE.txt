package main

import (
	"crypto/ecdh"
	"crypto/ecdsa"
	"crypto/elliptic"
	"encoding/base64"
	"encoding/json"
	"fmt"
	"math/big"

	jose "github.com/go-jose/go-jose/v4"
)

func b64(b []byte) string { return base64.RawURLEncoding.EncodeToString(b) }

func main() {
	d, _ := new(big.Int).SetString("c6ef9c5d78ae012a011164acb397ce2488a5d0d7a9a1f7ae0d7a8d5c1f5a1d37", 16)
	priv := new(ecdsa.PrivateKey)
	priv.Curve = elliptic.P256()
	priv.D = d
	priv.PublicKey.X, priv.PublicKey.Y = elliptic.P256().ScalarBaseMult(d.Bytes())
	ek, _ := priv.ECDH()
	_ = ecdh.P256
	pub := &priv.PublicKey
	body := `{"credentials":[{"credential":"eyJhbGciOiJFUzI1NiJ9.eyJ2Y3QiOiJ1cm46ZXVkaTpwaWQ6MSJ9.c2ln~disclosure~"}],"notification_id":"3fwe98js","credential_metadata":{"org.siros.secret_seed":{"org.siros.nym_auth":"00112233445566778899aabbccddeeff"}}}`
	out := map[string]any{
		"recipient_d": b64(ek.Bytes()),
		"recipient_x": b64(pub.X.FillBytes(make([]byte, 32))),
		"recipient_y": b64(pub.Y.FillBytes(make([]byte, 32))),
		"plaintext":   body,
	}
	cases := map[string]string{}
	for _, alg := range []jose.KeyAlgorithm{jose.ECDH_ES, jose.ECDH_ES_A128KW, jose.ECDH_ES_A192KW, jose.ECDH_ES_A256KW} {
		for _, enc := range []jose.ContentEncryption{jose.A128GCM, jose.A256GCM, jose.A128CBC_HS256, jose.A256CBC_HS512} {
			opts := (&jose.EncrypterOptions{}).WithContentType("JWT")
			enc2, err := jose.NewEncrypter(enc, jose.Recipient{Algorithm: alg, Key: pub}, opts)
			if err != nil {
				panic(err)
			}
			obj, err := enc2.Encrypt([]byte(body))
			if err != nil {
				panic(err)
			}
			s, _ := obj.CompactSerialize()
			cases[string(alg)+"|"+string(enc)] = s
		}
	}
	out["cases"] = cases
	j, _ := json.MarshalIndent(out, "", "  ")
	fmt.Println(string(j))
}
